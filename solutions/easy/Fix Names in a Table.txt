// Title: Fix Names in a Table
            // Difficulty: Easy
            // Language: MySQL
            // Link: https://leetcode.com/problems/fix-names-in-a-table/

# Write your MySQL query statement below

SELECT 
    user_id
    ,CONCAT(UPPER(SUBSTRING(name,1,1)),LOWER(SUBSTRING(name,2))) AS name
FROM users
ORDER BY user_id
